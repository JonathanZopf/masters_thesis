\chapter{Mathematical Demonstrations}
\label{app:mathematical_demonstrations}
This appendix provides supporting derivations for properties of \emph{FELICS} referenced in the main text. The demonstrations address the singleton special case, compatibility under reference-relative additivity, and the intended preservation of the dataset-average causal effect.

\section*{Equality of Causal Effect between FELICS singleton-groups and Original Metric}
\begin{math_box}[label=ma:one_item_group]
    Let the candidate interaction group consist of exactly one concept,
    \begin{equation}
        s=\{C_m\}.
    \end{equation}

    The payout coefficient introduced in Section~\ref{sec:shapley_correction} becomes

    \begin{equation}
        \gamma(\mathbf{x},s)
        =
        \frac{v_{\mathbf{x}}(s)}
        {\sum_{C_i\in s}v_{\mathbf{x}}(\{C_i\})}
        =
        \frac{v_{\mathbf{x}}(\{C_m\})}
        {v_{\mathbf{x}}(\{C_m\})}
        =1.
    \end{equation}

    Since the reference set used to compute the correction factor
    $\bar\gamma(\mathbf{x},s,T)$ is restricted to sets of identical treatment size,
    the only admissible reference size is likewise $|T|=1$. Consequently,

    \begin{equation}
        \bar\gamma(\mathbf{x},s,T)=1,
    \end{equation}

    and the normalized value function reduces to

    \begin{equation}
        v_{\mathrm{adj},\mathbf{x}}(T)
        =
        \frac{v_{\mathbf{x}}(T)}
        {\bar\gamma(\mathbf{x},s,T)}
        =
        v_{\mathbf{x}}(T).
    \end{equation}

    The adjusted causal effect is then obtained from the Shapley formulation

    \begin{equation}
        \mathrm{CE}_{\mathrm{adj}}(\mathbf{x},C_m,s)
        =
        \sum_{T\subseteq s\setminus\{C_m\}}
        \frac{|T|!(|s|-|T|-1)!}{|s|!}
        \left[
            v_{\mathrm{adj},\mathbf{x}}(T\cup\{C_m\})
            -
            v_{\mathrm{adj},\mathbf{x}}(T)
            \right].
    \end{equation}

    Since $s=\{C_m\}$, the only admissible coalition is the empty set,
    $T=\emptyset$. The weighting coefficient therefore simplifies to

    \begin{equation}
        \frac{0!(1-0-1)!}{1!}=1,
    \end{equation}

    and using the convention $v_{\mathrm{adj},\mathbf{x}}(\emptyset)=0$,

    \begin{equation}
        \mathrm{CE}_{\mathrm{adj}}(\mathbf{x},C_m,s)
        =
        v_{\mathrm{adj},\mathbf{x}}(\{C_m\}).
    \end{equation}

    Because no normalization is applied for singleton groups,

    \begin{equation}
        v_{\mathrm{adj},\mathbf{x}}(\{C_m\})
        =
        v_{\mathbf{x}}(\{C_m\}),
    \end{equation}

    which is exactly the value function underlying the causal-effect definition of Matton et al.~\cite{mattonWalkTalkMeasuring2024}. Therefore,

    \begin{equation}
        \mathrm{CE}_{\mathrm{adj}}(\mathbf{x},C_m,\{C_m\})
        =
        \mathrm{CE}_{\mathrm{Matton}}(\mathbf{x},C_m)
    \end{equation}

    showing that \emph{FELICS} reduces exactly to the original CE when every candidate interaction group contains only a single concept. Because the definitions of the explanation-implied effect and faithfulness (see Equations~\ref{eq:ee} and \ref{eq:faithfulness}) are unchanged, both metrics are equivalent in this scenario.
\end{math_box}

\section*{Compatibility under Reference-Relative Additivity}
\begin{math_box}[label=ma:independence_behavior]
    Assume a candidate interaction group $s$ that satisfies \emph{reference-relative
    additivity}: for every non-empty coalition $T\subseteq s$, its payout
    coefficient equals the singleton-weighted reference coefficient for treatments
    of the same size,

    \begin{equation}
        \bar\gamma(\mathbf{x},s,T)=\gamma(\mathbf{x},T)>\varepsilon,
        \qquad \forall\,\emptyset\neq T\subseteq s.
    \end{equation}

    This is a sufficient condition for the reference-relative additivity
    introduced in Section~\ref{sec:shapley_correction}; it is distinct from
    absolute additivity, which would require $\gamma(\mathbf{x},T)=1$. Under this
    assumption, the adjusted value of any non-empty coalition becomes

    \begin{equation}
        v_{\mathrm{adj},\mathbf{x}}(T)
        =
        \frac{v_{\mathbf{x}}(T)}
        {\gamma(\mathbf{x},T)}.
    \end{equation}

    Substituting the definition of the payout coefficient,

    \begin{equation}
        \gamma(\mathbf{x},T)
        =
        \frac{v_{\mathbf{x}}(T)}
        {\sum_{C_i\in T}v_{\mathbf{x}}(\{C_i\})},
    \end{equation}

    yields

    \begin{equation}
        v_{\mathrm{adj},\mathbf{x}}(T)
        =
        \sum_{C_i\in T}v_{\mathbf{x}}(\{C_i\}).
    \end{equation}

    Hence, after normalization, every coalition value equals the sum of its
    members' singleton intervention effects. The marginal contribution of a concept
    is therefore identical in every coalition in which it appears,

    \begin{equation}
        v_{\mathrm{adj},\mathbf{x}}(T\cup\{C_m\})
        -
        v_{\mathrm{adj},\mathbf{x}}(T)
        =
        v_{\mathbf{x}}(\{C_m\}),
        \qquad
        \forall\,T\subseteq s\setminus\{C_m\}.
    \end{equation}

    Since the Shapley weights form a partition of unity,

    \begin{equation}
        \sum_{T\subseteq s\setminus\{C_m\}}
        \frac{|T|!(|s|-|T|-1)!}{|s|!}
        =
        1,
    \end{equation}

    the adjusted causal effect reduces to

    \begin{equation}
        \mathrm{CE}_{\mathrm{adj}}(\mathbf{x},C_m,s)
        =
        v_{\mathbf{x}}(\{C_m\}).
    \end{equation}

    Finally, $v_{\mathbf{x}}(\{C_m\})$ is precisely the KL-divergence-based intervention value
    used by Matton et al.~\cite{mattonWalkTalkMeasuring2024} to define the causal
    effect of an individual concept (see Equation~\ref{eq:ce}). Consequently,

    \begin{equation}
        \mathrm{CE}_{\mathrm{adj}}(\mathbf{x},C_m,s)
        =
        \mathrm{CE}_{\mathrm{Matton}}(\mathbf{x},C_m),
    \end{equation}

    showing compatibility with the original metric for groups satisfying the stated
    reference-relative additivity assumptions.
\end{math_box}

\section*{CE Dataset-Average Preservation}
\begin{math_box}[label=ma:dataset_average_preservation]
    \textbf{Claim:} The singleton-weighted reference coefficient, formed as a
    ratio of aggregate joint effects to aggregate singleton effects, preserves the dataset-level average Causal Effect
    exactly when it is computed over the complete population of equally sized
    candidate interaction groups. A finite, target-excluding reference set provides an approximate
    estimator of this coefficient under the assumptions stated below. Consequently,
    the FELICS average is expected to approximate that of Matton et
    al.'s~\cite{mattonWalkTalkMeasuring2024} original metric:
    \begin{equation}
        \frac{1}{N_C}
        \sum_{\mathbf{x}\in\mathbf{X}}
        \sum_{s\in\mathbf{S}(\mathbf{x})}
        \sum_{C_m\in s}
        \mathrm{CE}_{\mathrm{adj}}(\mathbf{x},C_m,s)
        \approx
        \frac{1}{N_C}
        \sum_{\mathbf{x}\in\mathbf{X}}
        \sum_{C_m\in\mathbf{C}(\mathbf{x})}
        v_{\mathbf{x}}(\{C_m\}).
    \end{equation}
    Here, $N_C$ is the total number of concepts, $\mathbf{C}(\mathbf{x})$
    is the question-specific concept set, and $\mathbf{S}(\mathbf{x})$ is its
    partition into candidate interaction groups.

    \smallskip
    \textbf{Aggregate-Preserving Population Coefficient:} For a fixed treatment
    size $k$, define the population
    \begin{equation}
        \mathcal{G}_k
        :=
        \{(\mathbf{x},s)\mid \mathbf{x}\in\mathbf{X},\,
        s\in\mathbf{S}(\mathbf{x}),\,|s|=k\}.
    \end{equation}
    For each $q=(\mathbf{x},s)\in\mathcal{G}_k$, let
    \begin{equation}
        a_q:=v_{\mathbf{x}}(s),
        \qquad
        b_q:=\sum_{C_m\in s}v_{\mathbf{x}}(\{C_m\}),
        \qquad
        \gamma_q:=\frac{a_q}{b_q},
    \end{equation}
    where $b_q>0$. The demonstration concerns non-degenerate group sizes $k$ for
    which the coefficient defined below also satisfies $g_k>\varepsilon$. The
    size-specific coefficient that preserves the aggregate is
    \begin{equation}
        \label{eq:aggregate_preserving_gamma}
        g_k
        :=
        \frac{\sum_{q\in\mathcal{G}_k}a_q}
             {\sum_{q\in\mathcal{G}_k}b_q}
        =
        \frac{\sum_{q\in\mathcal{G}_k}b_q\gamma_q}
             {\sum_{q\in\mathcal{G}_k}b_q}.
    \end{equation}
    Thus, $g_k$ is a singleton-effect-weighted average of the payout
    coefficients, rather than their ordinary arithmetic mean.

    \smallskip
    \textbf{Exact Preservation for the Population Coefficient:} Suppose that
    every group $s$ of size $k$ is normalized using $g_k$. By Shapley efficiency,
    \begin{equation}
        \label{eq:shapley_efficiency_aggregate}
        \sum_{C_m\in s}
        \mathrm{CE}_{\mathrm{adj}}(\mathbf{x},C_m,s)
        =v_{\mathrm{adj},\mathbf{x}}(s)
        =\frac{v_{\mathbf{x}}(s)}{g_k}.
    \end{equation}
    Equation~\ref{eq:shapley_efficiency_aggregate} applies to every
    $(\mathbf{x},s)\in\mathcal{G}_k$. Summing both sides over this population
    therefore yields
    \begin{equation}
        \label{eq:aggregate_shapley_sum}
        \begin{aligned}
            &\sum_{(\mathbf{x},s)\in\mathcal{G}_k}
            \left[
                \sum_{C_m\in s}
                \mathrm{CE}_{\mathrm{adj}}(\mathbf{x},C_m,s)
            \right] \\
            &\qquad=
            \sum_{(\mathbf{x},s)\in\mathcal{G}_k}
            \frac{v_{\mathbf{x}}(s)}{g_k}
            =
            \frac{1}{g_k}
            \sum_{(\mathbf{x},s)\in\mathcal{G}_k}v_{\mathbf{x}}(s).
        \end{aligned}
    \end{equation}
    The last equality is valid because $g_k$ depends only on the common group
    size $k$ and is consequently constant with respect to the summation over
    $(\mathbf{x},s)$.

    To make the substitution from
    Equation~\ref{eq:aggregate_preserving_gamma} explicit, abbreviate the
    aggregate joint- and singleton-effect masses by
    \begin{equation*}
        A_k
        :=\sum_{(\mathbf{x},s)\in\mathcal{G}_k}v_{\mathbf{x}}(s),
        \qquad
        B_k
        :=\sum_{(\mathbf{x},s)\in\mathcal{G}_k}
        \sum_{C_m\in s}v_{\mathbf{x}}(\{C_m\}).
    \end{equation*}
    Equation~\ref{eq:aggregate_preserving_gamma} then states
    $g_k=A_k/B_k$. Since the non-degeneracy assumptions give $A_k>0$ and
    $B_k>0$, the right-hand side of
    Equation~\ref{eq:aggregate_shapley_sum} becomes
    \begin{equation}
        \label{eq:aggregate_preservation_size_k}
        \frac{1}{g_k}A_k
        =\frac{1}{A_k/B_k}A_k
        =\frac{B_k}{A_k}A_k
        =B_k
        =\sum_{(\mathbf{x},s)\in\mathcal{G}_k}
        \sum_{C_m\in s}v_{\mathbf{x}}(\{C_m\}).
    \end{equation}
    Combining Equations~\ref{eq:aggregate_shapley_sum} and
    \ref{eq:aggregate_preservation_size_k} proves exact aggregate preservation
    for groups of size $k$.

    Since $\mathbf{S}(\mathbf{x})$ partitions $\mathbf{C}(\mathbf{x})$,
    summing this identity over all treatment sizes and dividing by $N_C$ yields
    exact equality of the two dataset averages.

    \smallskip
    \textbf{Singleton-Weighted Reference Coefficient:} The finite-reference
    version of the coefficient is the following ratio of aggregates:
    \begin{equation}
        \label{eq:aggregate_reference_estimator}
        \bar\gamma(\mathbf{x},s,T)
        :=
        \frac{
            \displaystyle
            \sum_{(\mathbf{x}'',T'')\in\mathcal{R}(\mathbf{x},s,T)}
            v_{\mathbf{x}''}(T'')
        }{
            \displaystyle
            \sum_{(\mathbf{x}'',T'')\in\mathcal{R}(\mathbf{x},s,T)}
            \sum_{C_i\in T''}v_{\mathbf{x}''}(\{C_i\})
        }.
    \end{equation}
    For non-degenerate references, this is equivalent to weighting each payout
    coefficient by the corresponding treatment's aggregate singleton effect.

    \smallskip
    \textbf{Assumptions Made:}
    \begin{enumerate}
        \item For a target of size
        $k$, the reference set is representative of $\mathcal{G}_k$ with respect
        to both the joint effects $a_q$ and singleton-effect totals $b_q$.

        \item The aggregate singleton-effect
        denominator in Equation~\ref{eq:aggregate_reference_estimator} is positive
        and sufficiently far from zero, and the resulting coefficient exceeds
        $\varepsilon$ from Section~\ref{sec:corrected_value_function}. Degenerate size groups and reference sets invoke the
        unadjusted-value fallback and lie outside the preservation claim.

        \item The selected references provide enough
        coverage for their aggregate ratio to approximate the complete-population
        ratio $g_k$.
    \end{enumerate}
    Under these conditions,
    $\bar\gamma(\mathbf{x},s,s)\approx g_{|s|}$. Replacing $g_{|s|}$ in
    Equation~\ref{eq:shapley_efficiency_aggregate} with this target-specific
    estimate turns the exact population identity into an approximation.
\end{math_box}
