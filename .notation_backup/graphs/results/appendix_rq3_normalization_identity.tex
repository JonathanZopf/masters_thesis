% #1 data, #2 axis minimum, #3 axis maximum.
\newcommand{\NormalizationIdentityPlot}[3]{%
  \begin{minipage}[t]{\linewidth}
    \centering
    \begin{tikzpicture}
      \begin{axis}[
        width=5.55cm,
        height=5.55cm,
        scale only axis,
        xmin=#2,
        xmax=#3,
        ymin=#2,
        ymax=#3,
        xlabel={$\mathrm{CE}_{\mathrm{raw}}$},
        ylabel={$\mathrm{CE}_{\mathrm{adjusted}}$},
        scaled x ticks=false,
        scaled y ticks=false,
        tick label style={font=\scriptsize},
        label style={font=\scriptsize},
        grid=major,
        grid style={gray!18},
        axis line style={gray!60},
        clip=true,
      ]
        \addplot[
          only marks,
          mark=*,
          mark size=0.95pt,
          draw=NormalizationIdentityPoints,
          draw opacity=0.36,
          fill=NormalizationIdentityPoints,
          fill opacity=0.30,
        ] table[
          col sep=comma,
          x=raw_ce,
          y=adjusted_ce
        ] {#1};
        \addplot[
          NormalizationIdentityLine,
          very thick,
          dashed,
          domain=#2:#3,
          samples=2
        ] {x};
      \end{axis}
    \end{tikzpicture}
  \end{minipage}%
}
