% Look up the two posterior dataset-level estimates for a panel id.
% #1 panel id, #2 atomic result macro, #3 joint result macro.
\newcount\faithfulnessestimaterow
\newif\iffaithfulnessestimatefound
\newcommand{\FaithfulnessDatasetEstimates}[3]{%
  \global\faithfulnessestimatefoundfalse
  \pgfplotstablegetrowsof{\faithfulnessdatasetestimatetable}%
  \pgfmathtruncatemacro{\faithfulnessestimatelast}{\pgfplotsretval-1}%
  \faithfulnessestimaterow=0
  \loop\ifnum\faithfulnessestimaterow<\numexpr\faithfulnessestimatelast+1\relax
    \pgfplotstablegetelem{\the\faithfulnessestimaterow}{panel_id}\of\faithfulnessdatasetestimatetable
    \edef\faithfulnessestimatepanel{\pgfplotsretval}%
    \ifnum\pdfstrcmp{\faithfulnessestimatepanel}{#1}=0\relax
      \pgfplotstablegetelem{\the\faithfulnessestimaterow}{atomic_dataset_faithfulness}\of\faithfulnessdatasetestimatetable
      \xdef#2{\pgfplotsretval}%
      \pgfplotstablegetelem{\the\faithfulnessestimaterow}{joint_dataset_faithfulness}\of\faithfulnessdatasetestimatetable
      \xdef#3{\pgfplotsretval}%
      \global\faithfulnessestimatefoundtrue
    \fi
    \advance\faithfulnessestimaterow by 1
  \repeat
  \iffaithfulnessestimatefound\else
    \PackageError{felics-rainclouds}{No dataset estimate found for panel #1}%
      {Add the panel id and its estimates to \FaithfulnessDatasetEstimateFile.}%
  \fi
}

% tikzviolinplots computes each KDE in \violin@kde. Serialize its two relevant
% columns into an in-memory PGFPlots table before the next violin overwrites it.
\newcount\faithfulnesskderow
\makeatletter
\newcommand{\FaithfulnessCaptureCurrentKDE}[1]{%
  \pgfplotstablegetrowsof{\violin@kde}%
  \pgfmathtruncatemacro{\faithfulnesskdelast}{\pgfplotsretval-1}%
  \global\def#1{list kdecol\\}%
  \faithfulnesskderow=0
  \loop\ifnum\faithfulnesskderow<\numexpr\faithfulnesskdelast+1\relax
    \pgfplotstablegetelem{\the\faithfulnesskderow}{list}\of\violin@kde
    \edef\faithfulnesskdevalue{\pgfplotsretval}%
    \pgfplotstablegetelem{\the\faithfulnesskderow}{kdecol}\of\violin@kde
    \xdef#1{#1\faithfulnesskdevalue\space\pgfplotsretval\\}%
    \advance\faithfulnesskderow by 1
  \repeat
}
\newcommand{\FaithfulnessMaterializeKDE}[2]{%
  \begingroup
  \edef\faithfulnessloadtable{%
    \endgroup
    \noexpand\pgfplotstableread[row sep=\noexpand\\]%
      {\expandafter\unexpanded\expandafter{#1}}\noexpand#2}%
  \faithfulnessloadtable
}
\makeatother

% #1 project-local CSV, #2 file-safe panel id.
\newcommand{\FaithfulnessRaincloudPanel}[2]{%
  \begin{minipage}[t]{\linewidth}
  \centering
  % A CSV content change produces a new externalized filename. pgf/TikZ's
  % own MD5 check additionally invalidates the panel when its drawing code changes.
  \edef\FaithfulnessDataHash{\pdfmdfivesum file {#1}}%
  \edef\FaithfulnessEstimateHash{\pdfmdfivesum file {\FaithfulnessDatasetEstimateFile}}%
  \tikzsetnextfilename{faithfulness-#2-\FaithfulnessDataHash-\FaithfulnessEstimateHash}%
  \begin{tikzpicture}
    % Keep table processing inside the externalized picture. Cached panels
    % therefore skip both estimate lookup and KDE generation on later passes.
    \FaithfulnessDatasetEstimates{#2}%
      {\AtomicDatasetFaithfulness}{\JointDatasetFaithfulness}%
    % Use tikzviolinplots only for its native TeX KDE calculation.  The package
    % always draws a violin after computing \violin@kde, so clip those helper
    % axes to an empty region while retaining the computed table.
    \begin{scope}
    \clip (0,0) rectangle (0,0);
    \violinsetoptions[scaled,reverseaxis]{
      width=7.8cm,
      height=5.40cm,
      overlay,
      xmin=-1,
      xmax=1,
      ymin=.10,
      ymax=7.25,
      axis lines=none,
    }
    \violinplot[col sep=comma,index=atomic_faithfulness,relative position=5.05,color=white,label={},samples=30]{#1}
    \FaithfulnessCaptureCurrentKDE{\AtomicFaithfulnessKDE}
    \violinplot[col sep=comma,index=interaction_faithfulness,relative position=1.65,color=white,label={},samples=30]{#1}
    \FaithfulnessCaptureCurrentKDE{\JointFaithfulnessKDE}
    \end{scope}
    \FaithfulnessMaterializeKDE{\AtomicFaithfulnessKDE}{\AtomicFaithfulnessKDETable}
    \FaithfulnessMaterializeKDE{\JointFaithfulnessKDE}{\JointFaithfulnessKDETable}
    
    \pgfresetboundingbox
    \begin{axis}[
      width=7.8cm,
      height=5.55cm,
      xmin=-1,
      xmax=1,
      ymin=.10,
      ymax=7.25,
      axis x line*=bottom,
      axis y line=none,
      xtick={-1,-.5,0,.5,1},
      xmajorgrids=true,
      grid style={gray!18},
      % Faithfulness is bounded by [-1,1]. Clipping prevents the KDE tails
      % from enlarging a panel or suggesting probability outside that domain.
      clip=true,
    ]
      \addplot[gray!55,densely dashed,thin]
        coordinates {(0,.42) (0,6.35)};
      \addplot[draw=FelicsAtomic,fill=FelicsAtomic,fill opacity=.48]
        table[x=list,y expr=5.15+1.6*(\thisrow{kdecol}-5.05)]
          {\AtomicFaithfulnessKDETable};
      \addplot[only marks,mark=*,mark size=1.15pt,
        draw=FelicsAtomic,fill=FelicsAtomic,fill opacity=.62,
        y filter/.expression={4.35+.16*(mod(\coordindex*7,11)/10-.5)}]
        table[col sep=comma,x=atomic_faithfulness,y expr=4.35]
          {#1};
      \addplot+[boxplot={draw direction=x,draw position=4.02},
        mark=none,draw=FelicsAtomic,fill=FelicsAtomic,fill opacity=.32,
        thick,boxplot/box extend=.18]
        table[col sep=comma,y=atomic_faithfulness]
          {#1};
      \addplot[draw=FelicsJoint,fill=FelicsJoint,fill opacity=.48]
        table[x=list,y expr=1.75+1.6*(\thisrow{kdecol}-1.65)]
          {\JointFaithfulnessKDETable};
      \addplot[only marks,mark=*,mark size=1.15pt,
        draw=FelicsJoint,fill=FelicsJoint,fill opacity=.66,
        y filter/.expression={.92+.16*(mod(\coordindex*7,11)/10-.5)}]
        table[col sep=comma,x=interaction_faithfulness,y expr=.92]
          {#1};
      \addplot+[boxplot={draw direction=x,draw position=.55},
        mark=none,draw=FelicsJoint,fill=FelicsJoint,fill opacity=.34,
        thick,solid,boxplot/box extend=.18]
        table[col sep=comma,y=interaction_faithfulness]
          {#1};
      \node[anchor=west,font=\scriptsize\bfseries,text=FelicsAtomic]
        at (axis cs:-.95,6.80) {Atomic};
      \node[anchor=west,font=\scriptsize,text=black]
        at (axis cs:-.55,6.80)
        {$\mathcal{F}(\mathbf{X})=
          \pgfmathprintnumber[fixed,fixed zerofill,precision=3]{\AtomicDatasetFaithfulness}$};
      \node[anchor=west,font=\scriptsize\bfseries,text=FelicsJoint]
        at (axis cs:-.95,3.40) {Joint};
      \node[anchor=west,font=\scriptsize,text=black]
        at (axis cs:-.55,3.40)
        {$\mathcal{F}(\mathbf{X})=
          \pgfmathprintnumber[fixed,fixed zerofill,precision=3]{\JointDatasetFaithfulness}$};
    \end{axis}
  \end{tikzpicture}%
  \end{minipage}%
}
