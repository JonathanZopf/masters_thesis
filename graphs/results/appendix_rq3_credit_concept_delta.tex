\definecolor{CreditDeltaBars}{HTML}{2878B5}

% Mean normalization adjustment per New German Credit concept. Concepts are
% ordered by their oracle group, and dotted lines separate the five groups.
% #1 data file.
\newcommand{\CreditConceptDeltaPlot}[1]{%
  \begin{tikzpicture}
    \begin{axis}[
      % The axis excludes the long concept labels, which take about 3.75cm.
      scale only axis,
      width=\linewidth-3.75cm,
      height=5.3cm,
      xbar,
      bar width=5pt,
      xmin=0,
      xmax=0.4,
      xtick={0,0.1,0.2,0.3,0.4},
      ymin=0.4,
      ymax=12.6,
      y dir=reverse,
      ytick={1,...,12},
      yticklabels={
        Credit Purpose,
        Credit Amount,
        Credit Term (months),
        Liquid Assets,
        Household Income,
        SCHUFA Score (\%),
        Real Estate Value,
        Applicant Age,
        Lives In Own House,
        Monthly Debt Burden,
        Job Category,
        Household Size
      },
      xlabel={$\overline{\Delta\mathrm{CE}}=\overline{\mathrm{CE}_{\mathrm{adjusted}}-\mathrm{CE}_{\mathrm{raw}}}$},
      scaled x ticks=false,
      xticklabel style={font=\scriptsize,/pgf/number format/fixed,/pgf/number format/precision=1},
      yticklabel style={font=\scriptsize},
      label style={font=\scriptsize},
      xmajorgrids,
      grid style={gray!18},
      axis line style={gray!60},
      axis y line*=left,
      axis x line*=bottom,
      clip=false,
    ]
      \foreach \groupborder in {3.5,6.5,9.5,11.5}{
        \edef\temp{\noexpand\draw[gray!55,densely dotted]
          (axis cs:0,\groupborder) -- (axis cs:0.4,\groupborder);}
        \temp
      }
      \addplot[
        CreditDeltaBars,
        fill=CreditDeltaBars,
        fill opacity=0.72,
        draw=CreditDeltaBars,
      ] table[
        col sep=comma,
        x=mean_delta_ce,
        y=concept_id
      ] {#1};
    \end{axis}
  \end{tikzpicture}%
}
