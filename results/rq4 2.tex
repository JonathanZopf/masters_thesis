% #1 title, #2 data, #3 axis minimum, #4 axis maximum, #5 sample size.
\newcommand{\HypFourIdentityPlot}[5]{%
  \begin{minipage}[t]{0.47\textwidth}
    \centering
    {\small\bfseries #1\par}
    \vspace{0.1em}
    \begin{tikzpicture}
      \begin{axis}[
        width=5.55cm,
        height=5.55cm,
        scale only axis,
        xmin=#3,
        xmax=#4,
        ymin=#3,
        ymax=#4,
        xlabel={$\mathcal{F}_{\mathrm{atomic}}(\mathbf{x})$},
        ylabel={$\mathcal{F}_{\mathrm{joint}}(\mathbf{x})$},
        tick label style={font=\scriptsize},
        label style={font=\scriptsize},
        grid=major,
        grid style={gray!18},
        axis line style={gray!60},
      ]
        \addplot[
          only marks,mark=*,mark size=1.8pt,
          draw=HypFourPoints,fill=HypFourPoints,fill opacity=0.68,
        ] table[col sep=comma,x=atomic_faithfulness,y=joint_faithfulness] {#2};
        \addplot[HypFourIdentity,very thick,dashed,domain=#3:#4,samples=2] {x};
      \end{axis}
    \end{tikzpicture}
    \vspace{-0.2em}
    {\scriptsize Paired questions: $n=#5$\par}
  \end{minipage}%
}
